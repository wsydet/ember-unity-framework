await frames(30);
var timings=new FrameTiming[1];var rows=new System.Collections.Generic.List<object>();
var main=new System.Collections.Generic.List<double>();var cpu=new System.Collections.Generic.List<double>();var gpu=new System.Collections.Generic.List<double>();var render=new System.Collections.Generic.List<double>();
var draw=Unity.Profiling.ProfilerRecorder.StartNew(Unity.Profiling.ProfilerCategory.Render,"Draw Calls Count",1);
var gc=Unity.Profiling.ProfilerRecorder.StartNew(Unity.Profiling.ProfilerCategory.Memory,"GC Allocated In Frame",1);
var host=Ember.UIRendering.EUIBackdropCameraHost.Active;int begin=host.CaptureCount;ulong previous=0;
for(int i=0;i<180;i++){
FrameTimingManager.CaptureFrameTimings();await frames(1);
if(FrameTimingManager.GetLatestTimings(1,timings)>0&&timings[0].frameStartTimestamp!=previous){
var t=timings[0];previous=t.frameStartTimestamp;cpu.Add(t.cpuFrameTime);main.Add(t.cpuMainThreadFrameTime);render.Add(t.cpuRenderThreadFrameTime);if(t.gpuFrameTime>0)gpu.Add(t.gpuFrameTime);
rows.Add(new{cpu=t.cpuFrameTime,main=t.cpuMainThreadFrameTime,render=t.cpuRenderThreadFrameTime,gpu=t.gpuFrameTime,draw=draw.Valid?draw.LastValue:-1,gcBytes=gc.Valid?gc.LastValue:-1,stamp=t.frameStartTimestamp});
}}
System.Func<System.Collections.Generic.List<double>,object> stats=list=>{list.Sort();return new{count=list.Count,mean=list.Count>0?list.Average():0,p50=list.Count>0?list[list.Count/2]:0,p95=list.Count>0?list[(int)((list.Count-1)*.95)]:0};};
var rt=Resources.FindObjectsOfTypeAll<RenderTexture>().Where(r=>r.name.StartsWith("EUI Backdrop Output")&&r.IsCreated()).ToArray();
output.Add(new{label=label,warmup=30,frames=180,screen=new[]{Screen.width,Screen.height},boundaries=host.BoundaryCount,captures=host.CaptureCount-begin,rt=rt.Select(r=>new{w=r.width,h=r.height,format=r.graphicsFormat.ToString(),bytes=UnityEngine.Profiling.Profiler.GetRuntimeMemorySizeLong(r)}).ToArray(),cpu=stats(cpu),main=stats(main),render=stats(render),gpu=stats(gpu),rows=rows});
draw.Dispose();gc.Dispose();System.IO.File.WriteAllText(path,Newtonsoft.Json.JsonConvert.SerializeObject(output,Newtonsoft.Json.Formatting.Indented));
};
